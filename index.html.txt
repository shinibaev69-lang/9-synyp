    let currentVariant = 1;
    let userAnswers = {};

    // Бет жүктелгенде тестті шығару
    window.onload = function() {
        renderQuiz();
    };

    // Нұсқаны ауыстыру
    function selectVariant(variant) {
        currentVariant = variant;
        
        // Түймелердің белсенділігін өзгерту
        const buttons = document.querySelectorAll('.variant-select .btn');
        buttons.forEach((btn, idx) => {
            if (idx + 1 === variant) {
                btn.classList.add('active');
            } else {
                btn.classList.remove('active');
            }
        });

        resetQuiz();
    }

    // Тестті экранға шығару
    function renderQuiz() {
        const container = document.getElementById('quiz-container');
        container.innerHTML = '';
        const questions = quizData[currentVariant];

        questions.forEach((q, qIndex) => {
            const card = document.createElement('div');
            card.className = 'question-card';
            card.id = `q-card-${qIndex}`;

            let optionsHTML = '';
            q.options.forEach((opt, oIndex) => {
                optionsHTML += `
                    <label class="option-label" id="label-${qIndex}-${oIndex}">
                        <input type="radio" name="question-${qIndex}" value="${oIndex}" onchange="registerAnswer(${qIndex}, ${oIndex})">
                        ${opt}
                    </label>
                `;
            });

            card.innerHTML = `
                <div class="learning-objective">Оқу мақсаты: ${q.objective}</div>
                <div class="question-text">${q.question}</div>
                <div class="options">${optionsHTML}</div>
                <div class="feedback" id="feedback-${qIndex}"></div>
            `;

            container.appendChild(card);
        });
    }

    // Тұтынушының жауабын белгілеу және оған бірден смайлик көрсету
    function registerAnswer(qIndex, selectedOption) {
        userAnswers[qIndex] = selectedOption;
        const correctOption = quizData[currentVariant][qIndex].correct;
        const feedbackDiv = document.getElementById(`feedback-${qIndex}`);

        // Жауап берілсе, бірден тексеріп смайлик көрсету
        feedbackDiv.style.display = 'flex';
        if (selectedOption === correctOption) {
            feedbackDiv.innerHTML = `<span>😊 Өте жақсы! Дұрыс жауап!</span>`;
            feedbackDiv.style.color = '#27ae60';
        } else {
            feedbackDiv.innerHTML = `<span>😔 Әттеген-ай, қате! Дұрыс жауабын ойланып көріңіз.</span>`;
            feedbackDiv.style.color = '#e74c3c';
        }
    }

    // Барлық жауаптарды тексеру және соңғы нәтижені шығару
    function checkAnswers() {
        const questions = quizData[currentVariant];
        let score = 0;

        questions.forEach((q, qIndex) => {
            const selected = userAnswers[qIndex];
            const feedbackDiv = document.getElementById(`feedback-${qIndex}`);

            q.options.forEach((_, oIndex) => {
                const label = document.getElementById(`label-${qIndex}-${oIndex}`);
                label.classList.remove('correct-bg', 'wrong-bg');

                if (oIndex === q.correct) {
                    label.classList.add('correct-bg'); // Дұрыс жауапты жасылмен белгілеу
                }
                if (selected === oIndex && selected !== q.correct) {
                    label.classList.add('wrong-bg'); // Қате жауапты қызылмен белгілеу
                }
            });

            if (selected === q.correct) {
                score++;
            }
        });

        // Соңғы нәтижені көрсету
        const resultBox = document.getElementById('result-box');
        const scoreText = document.getElementById('final-score');
        const emojiDiv = document.getElementById('final-emoji');

        resultBox.style.display = 'block';
        scoreText.innerText = `Сіз 10 сұрақтың ${score} сұрағына дұрыс жауап бердіңіз!`;

        if (score >= 9) {
            emojiDiv.innerText = "🏆 🎉 🌟";
            scoreText.innerHTML += "<br><strong>Өте үздік нәтиже! Жарайсың!</strong>";
        } else if (score >= 7) {
            emojiDiv.innerText = "👍 😊 ✨";
            scoreText.innerHTML += "<br><strong>Жақсы нәтиже! Тағы да іздене түс!</strong>";
        } else if (score >= 5) {
            emojiDiv.innerText = "😐 📚";
            scoreText.innerHTML += "<br><strong>Орташа нәтиже. Тақырыпты қайталап шығу ұсынылады.</strong>";
        } else {
            emojiDiv.innerText = "😢 📖 💡";
            scoreText.innerHTML += "<br><strong>Төмен нәтиже. Материалды қайта оқып тапсыруды кеңес береміз!</strong>";
        }

        // Бетті нәтиже тұсына ақырын айналдыру
        resultBox.scrollIntoView({ behavior: 'smooth' });
    }

    // Тестті қайта бастау
    function resetQuiz() {
        userAnswers = {};
        document.getElementById('result-box').style.display = 'none';
        renderQuiz();
        window.scrollTo({ top: 0, behavior: 'smooth' });
    }
